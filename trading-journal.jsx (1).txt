import { useState, useEffect } from "react";

const STORAGE_KEY = "jericho-trade-journal";

const defaultTrade = {
  date: "",
  pair: "",
  direction: "LONG",
  entry: "",
  sl: "",
  tp: "",
  exitPrice: "",
  lotSize: "",
  result: "WIN",
  pnl: "",
  rr: "",
  setup: "",
  notes: "",
};

const PAIRS = ["EUR/USD", "USD/JPY", "GBP/USD", "USD/CHF", "AUD/USD", "NZD/USD", "USD/CAD", "Other"];
const SETUPS = ["BOS", "CHOCH", "FVG", "OB", "Liquidity Sweep", "Fib", "Structure", "Other"];

export default function TradingJournal() {
  const [trades, setTrades] = useState([]);
  const [form, setForm] = useState(defaultTrade);
  const [view, setView] = useState("dashboard"); // dashboard | log | add
  const [editId, setEditId] = useState(null);
  const [filter, setFilter] = useState("ALL");

  useEffect(() => {
    (async () => {
      try {
        const res = await window.storage.get(STORAGE_KEY);
        if (res) setTrades(JSON.parse(res.value));
      } catch {}
    })();
  }, []);

  const persist = async (updated) => {
    await window.storage.set(STORAGE_KEY, JSON.stringify(updated));
    setTrades(updated);
  };

  const calcStats = (list) => {
    if (!list.length) return { total: 0, wins: 0, losses: 0, wr: 0, totalPnl: 0, avgRR: 0, bestWin: 0, worstLoss: 0 };
    const wins = list.filter(t => t.result === "WIN");
    const losses = list.filter(t => t.result === "LOSS");
    const totalPnl = list.reduce((s, t) => s + (parseFloat(t.pnl) || 0), 0);
    const avgRR = list.reduce((s, t) => s + (parseFloat(t.rr) || 0), 0) / list.length;
    const pnls = list.map(t => parseFloat(t.pnl) || 0);
    return {
      total: list.length,
      wins: wins.length,
      losses: losses.length,
      wr: ((wins.length / list.length) * 100).toFixed(1),
      totalPnl: totalPnl.toFixed(2),
      avgRR: avgRR.toFixed(2),
      bestWin: Math.max(0, ...pnls).toFixed(2),
      worstLoss: Math.min(0, ...pnls).toFixed(2),
    };
  };

  const filtered = filter === "ALL" ? trades : trades.filter(t => t.result === filter);
  const stats = calcStats(trades);

  const handleSubmit = async () => {
    if (!form.date || !form.pair || !form.entry) return;
    if (editId !== null) {
      const updated = trades.map(t => t.id === editId ? { ...form, id: editId } : t);
      await persist(updated);
      setEditId(null);
    } else {
      const newTrade = { ...form, id: Date.now() };
      await persist([newTrade, ...trades]);
    }
    setForm(defaultTrade);
    setView("log");
  };

  const deleteTrade = async (id) => {
    await persist(trades.filter(t => t.id !== id));
  };

  const startEdit = (trade) => {
    setForm(trade);
    setEditId(trade.id);
    setView("add");
  };

  const pnlColor = (val) => {
    const n = parseFloat(val);
    if (n > 0) return "#00e5a0";
    if (n < 0) return "#ff4d6d";
    return "#888";
  };

  const inp = `
    background: #0d0d0d;
    border: 1px solid #2a2a2a;
    border-radius: 6px;
    color: #e0e0e0;
    padding: 8px 12px;
    font-size: 13px;
    width: 100%;
    outline: none;
    box-sizing: border-box;
    font-family: 'JetBrains Mono', monospace;
  `;

  return (
    <div style={{
      minHeight: "100vh",
      background: "#080808",
      color: "#e0e0e0",
      fontFamily: "'JetBrains Mono', 'Courier New', monospace",
      padding: "0",
    }}>
      {/* Header */}
      <div style={{
        borderBottom: "1px solid #1a1a1a",
        padding: "18px 28px",
        display: "flex",
        alignItems: "center",
        justifyContent: "space-between",
        background: "#0a0a0a",
      }}>
        <div>
          <div style={{ fontSize: "11px", color: "#555", letterSpacing: "3px", textTransform: "uppercase" }}>Jericho</div>
          <div style={{ fontSize: "20px", fontWeight: "700", color: "#fff", letterSpacing: "1px" }}>Trade Journal</div>
        </div>
        <div style={{ display: "flex", gap: "8px" }}>
          {["dashboard", "log", "add"].map(v => (
            <button key={v} onClick={() => { setView(v); if (v === "add") { setForm(defaultTrade); setEditId(null); } }}
              style={{
                background: view === v ? "#00e5a0" : "transparent",
                color: view === v ? "#000" : "#555",
                border: `1px solid ${view === v ? "#00e5a0" : "#2a2a2a"}`,
                borderRadius: "5px",
                padding: "6px 14px",
                fontSize: "11px",
                fontFamily: "inherit",
                cursor: "pointer",
                textTransform: "uppercase",
                letterSpacing: "1px",
                fontWeight: "600",
                transition: "all 0.2s",
              }}>
              {v === "add" ? (editId ? "Edit" : "+ New") : v}
            </button>
          ))}
        </div>
      </div>

      <div style={{ padding: "24px 28px", maxWidth: "1100px", margin: "0 auto" }}>

        {/* DASHBOARD */}
        {view === "dashboard" && (
          <div>
            <div style={{ display: "grid", gridTemplateColumns: "repeat(auto-fit, minmax(160px, 1fr))", gap: "12px", marginBottom: "28px" }}>
              {[
                { label: "Total Trades", val: stats.total, accent: "#fff" },
                { label: "Win Rate", val: `${stats.wr}%`, accent: parseFloat(stats.wr) >= 50 ? "#00e5a0" : "#ff4d6d" },
                { label: "Total P&L", val: `$${stats.totalPnl}`, accent: pnlColor(stats.totalPnl) },
                { label: "Wins", val: stats.wins, accent: "#00e5a0" },
                { label: "Losses", val: stats.losses, accent: "#ff4d6d" },
                { label: "Avg R:R", val: stats.avgRR, accent: "#a78bfa" },
                { label: "Best Win", val: `$${stats.bestWin}`, accent: "#00e5a0" },
                { label: "Worst Loss", val: `$${stats.worstLoss}`, accent: "#ff4d6d" },
              ].map(s => (
                <div key={s.label} style={{
                  background: "#0f0f0f",
                  border: "1px solid #1a1a1a",
                  borderRadius: "8px",
                  padding: "16px 18px",
                }}>
                  <div style={{ fontSize: "10px", color: "#555", letterSpacing: "2px", textTransform: "uppercase", marginBottom: "8px" }}>{s.label}</div>
                  <div style={{ fontSize: "22px", fontWeight: "700", color: s.accent }}>{s.val}</div>
                </div>
              ))}
            </div>

            {/* Equity line (simple) */}
            {trades.length > 1 && (() => {
              let running = 0;
              const points = [...trades].reverse().map((t, i) => {
                running += parseFloat(t.pnl) || 0;
                return { i, v: running };
              });
              const min = Math.min(...points.map(p => p.v));
              const max = Math.max(...points.map(p => p.v));
              const range = max - min || 1;
              const W = 700, H = 120;
              const x = (i) => (i / (points.length - 1)) * W;
              const y = (v) => H - ((v - min) / range) * H;
              const path = points.map((p, i) => `${i === 0 ? "M" : "L"} ${x(p.i)} ${y(p.v)}`).join(" ");
              const fill = points.map((p, i) => `${i === 0 ? "M" : "L"} ${x(p.i)} ${y(p.v)}`).join(" ") + ` L ${x(points.length - 1)} ${H} L 0 ${H} Z`;
              return (
                <div style={{ background: "#0f0f0f", border: "1px solid #1a1a1a", borderRadius: "8px", padding: "18px", marginBottom: "20px" }}>
                  <div style={{ fontSize: "10px", color: "#555", letterSpacing: "2px", textTransform: "uppercase", marginBottom: "12px" }}>Equity Curve</div>
                  <svg viewBox={`0 0 ${W} ${H}`} style={{ width: "100%", height: "100px", display: "block" }}>
                    <defs>
                      <linearGradient id="eg" x1="0" y1="0" x2="0" y2="1">
                        <stop offset="0%" stopColor="#00e5a0" stopOpacity="0.25" />
                        <stop offset="100%" stopColor="#00e5a0" stopOpacity="0" />
                      </linearGradient>
                    </defs>
                    <path d={fill} fill="url(#eg)" />
                    <path d={path} fill="none" stroke="#00e5a0" strokeWidth="1.5" />
                  </svg>
                </div>
              );
            })()}

            {trades.length === 0 && (
              <div style={{ textAlign: "center", color: "#333", padding: "60px 0", fontSize: "13px" }}>
                No trades logged yet. Hit <span style={{ color: "#00e5a0" }}>+ New</span> to add your first one.
              </div>
            )}
          </div>
        )}

        {/* LOG */}
        {view === "log" && (
          <div>
            <div style={{ display: "flex", gap: "8px", marginBottom: "20px", flexWrap: "wrap" }}>
              {["ALL", "WIN", "LOSS", "BE"].map(f => (
                <button key={f} onClick={() => setFilter(f)} style={{
                  background: filter === f ? "#1a1a1a" : "transparent",
                  border: `1px solid ${filter === f ? "#00e5a0" : "#2a2a2a"}`,
                  color: filter === f ? "#00e5a0" : "#444",
                  borderRadius: "5px",
                  padding: "5px 12px",
                  fontSize: "11px",
                  fontFamily: "inherit",
                  cursor: "pointer",
                  letterSpacing: "1px",
                }}>{f}</button>
              ))}
              <div style={{ marginLeft: "auto", color: "#444", fontSize: "12px", alignSelf: "center" }}>
                {filtered.length} trade{filtered.length !== 1 ? "s" : ""}
              </div>
            </div>
            {filtered.length === 0 ? (
              <div style={{ textAlign: "center", color: "#333", padding: "60px 0", fontSize: "13px" }}>Nothing here yet.</div>
            ) : (
              <div style={{ display: "flex", flexDirection: "column", gap: "8px" }}>
                {filtered.map(t => (
                  <div key={t.id} style={{
                    background: "#0f0f0f",
                    border: `1px solid #1c1c1c`,
                    borderLeft: `3px solid ${t.result === "WIN" ? "#00e5a0" : t.result === "LOSS" ? "#ff4d6d" : "#555"}`,
                    borderRadius: "8px",
                    padding: "14px 18px",
                    display: "grid",
                    gridTemplateColumns: "80px 90px 60px 1fr 80px 80px 60px auto",
                    gap: "12px",
                    alignItems: "center",
                    fontSize: "12px",
                  }}>
                    <div style={{ color: "#555" }}>{t.date}</div>
                    <div style={{ color: "#a78bfa", fontWeight: "600" }}>{t.pair}</div>
                    <div style={{ color: t.direction === "LONG" ? "#00e5a0" : "#ff4d6d", fontSize: "10px", letterSpacing: "1px" }}>{t.direction}</div>
                    <div style={{ color: "#444", fontSize: "11px", overflow: "hidden", textOverflow: "ellipsis", whiteSpace: "nowrap" }}>{t.setup || "—"}</div>
                    <div style={{ color: pnlColor(t.pnl), fontWeight: "700" }}>{t.pnl ? `$${parseFloat(t.pnl).toFixed(2)}` : "—"}</div>
                    <div style={{ color: "#555" }}>{t.rr ? `${t.rr}R` : "—"}</div>
                    <div style={{ color: t.result === "WIN" ? "#00e5a0" : t.result === "LOSS" ? "#ff4d6d" : "#888", fontSize: "10px", letterSpacing: "1px", fontWeight: "700" }}>{t.result}</div>
                    <div style={{ display: "flex", gap: "6px" }}>
                      <button onClick={() => startEdit(t)} style={{ background: "none", border: "1px solid #2a2a2a", color: "#555", borderRadius: "4px", padding: "3px 8px", cursor: "pointer", fontSize: "11px", fontFamily: "inherit" }}>edit</button>
                      <button onClick={() => deleteTrade(t.id)} style={{ background: "none", border: "1px solid #2a2a2a", color: "#ff4d6d", borderRadius: "4px", padding: "3px 8px", cursor: "pointer", fontSize: "11px", fontFamily: "inherit" }}>×</button>
                    </div>
                  </div>
                ))}
              </div>
            )}
          </div>
        )}

        {/* ADD / EDIT */}
        {view === "add" && (
          <div style={{ maxWidth: "640px" }}>
            <div style={{ fontSize: "11px", color: "#555", letterSpacing: "3px", textTransform: "uppercase", marginBottom: "20px" }}>
              {editId ? "Edit Trade" : "Log Trade"}
            </div>
            <div style={{ display: "grid", gridTemplateColumns: "1fr 1fr", gap: "14px" }}>
              {[
                { label: "Date", key: "date", type: "date" },
                { label: "Pair", key: "pair", type: "select", opts: PAIRS },
                { label: "Direction", key: "direction", type: "select", opts: ["LONG", "SHORT"] },
                { label: "Setup", key: "setup", type: "select", opts: SETUPS },
                { label: "Entry", key: "entry", type: "number", placeholder: "1.08500" },
                { label: "SL", key: "sl", type: "number", placeholder: "1.08200" },
                { label: "TP", key: "tp", type: "number", placeholder: "1.09100" },
                { label: "Exit Price", key: "exitPrice", type: "number", placeholder: "1.09050" },
                { label: "Lot Size", key: "lotSize", type: "number", placeholder: "0.01" },
                { label: "P&L ($)", key: "pnl", type: "number", placeholder: "+25.00" },
                { label: "R:R", key: "rr", type: "number", placeholder: "1.5" },
                { label: "Result", key: "result", type: "select", opts: ["WIN", "LOSS", "BE"] },
              ].map(f => (
                <div key={f.key}>
                  <div style={{ fontSize: "10px", color: "#555", letterSpacing: "2px", textTransform: "uppercase", marginBottom: "6px" }}>{f.label}</div>
                  {f.type === "select" ? (
                    <select value={form[f.key]} onChange={e => setForm({ ...form, [f.key]: e.target.value })}
                      style={{ ...Object.fromEntries(inp.split(";").filter(Boolean).map(s => { const [k, v] = s.split(":"); return [k.trim().replace(/-([a-z])/g, g => g[1].toUpperCase()), v?.trim()]; })) }}>
                      {f.opts.map(o => <option key={o} value={o}>{o}</option>)}
                    </select>
                  ) : (
                    <input type={f.type} value={form[f.key]} placeholder={f.placeholder || ""}
                      onChange={e => setForm({ ...form, [f.key]: e.target.value })}
                      style={{ background: "#0d0d0d", border: "1px solid #2a2a2a", borderRadius: "6px", color: "#e0e0e0", padding: "8px 12px", fontSize: "13px", width: "100%", outline: "none", boxSizing: "border-box", fontFamily: "inherit" }} />
                  )}
                </div>
              ))}
            </div>
            <div style={{ marginTop: "14px" }}>
              <div style={{ fontSize: "10px", color: "#555", letterSpacing: "2px", textTransform: "uppercase", marginBottom: "6px" }}>Notes</div>
              <textarea value={form.notes} onChange={e => setForm({ ...form, notes: e.target.value })} rows={3} placeholder="What did you see? What did you feel? What would you do differently?"
                style={{ background: "#0d0d0d", border: "1px solid #2a2a2a", borderRadius: "6px", color: "#e0e0e0", padding: "10px 12px", fontSize: "13px", width: "100%", outline: "none", boxSizing: "border-box", fontFamily: "inherit", resize: "vertical" }} />
            </div>
            <div style={{ display: "flex", gap: "10px", marginTop: "20px" }}>
              <button onClick={handleSubmit} style={{
                background: "#00e5a0", color: "#000", border: "none", borderRadius: "6px", padding: "10px 28px",
                fontSize: "12px", fontFamily: "inherit", fontWeight: "700", letterSpacing: "1px", cursor: "pointer", textTransform: "uppercase",
              }}>
                {editId ? "Update" : "Save Trade"}
              </button>
              <button onClick={() => { setForm(defaultTrade); setEditId(null); setView("log"); }} style={{
                background: "transparent", color: "#555", border: "1px solid #2a2a2a", borderRadius: "6px", padding: "10px 20px",
                fontSize: "12px", fontFamily: "inherit", cursor: "pointer", textTransform: "uppercase", letterSpacing: "1px",
              }}>Cancel</button>
            </div>
          </div>
        )}
      </div>
    </div>
  );
}
